const MAP = {
  "USD": "USD",
  "ECU": "EUR",
  "EUR": "EUR",
  "MLC": "MLC"
};

let chart = null;
let latestRates = { USD: null, EUR: null, MLC: null };

function actualizarFraseDolarDesdeTasas(data){
  try{
    const usd = data?.data?.tasas?.USD;
    if(usd === undefined) return;

    const now = new Date();
    const yyyy = now.getFullYear();
    const mm = String(now.getMonth()+1).padStart(2,'0');
    const dd = String(now.getDate()).padStart(2,'0');
    const fechaIso = `${yyyy}-${mm}-${dd}`;

    const n = Number(usd);
    const el = document.getElementById("qaDolar");
    if(!el) return;

    el.innerHTML =
      `Hoy <strong>${fechaIso}</strong> el <strong>dólar (USD)</strong> en Cuba tiene un valor de ` +
      `<strong>${n.toFixed(2)} CUP</strong> como referencia informativa del mercado informal. ` +
      `La tabla se actualiza automáticamente y muestra la hora de la última actualización.`;
  }catch(e){}
}

async function cargarTasas(){
  try {
    const r = await fetch("tasas.php", { cache: "no-store" });
    const data = await r.json();

    if (!data || !data.data || !data.data.tasas) {
      console.error("Respuesta de tasas.php no tiene el formato esperado", data);
      return;
    }

    actualizarFraseDolarDesdeTasas(data);

    const tasas = data.data.tasas;
    const old = JSON.parse(localStorage.getItem("tasas_old") || "{}");
    const filas = document.querySelectorAll("#mercado-informal .rate-row[data-code]");
    const nuevosValores = [];

    filas.forEach(fila => {
      const code = fila.dataset.code;
      const apiKey = Object.keys(MAP).find(k => MAP[k] === code);

      if (apiKey && tasas[apiKey] !== undefined) {
        const nuevo = parseFloat(tasas[apiKey]);
        const anterior = (old[apiKey] ?? nuevo);
        const diff = nuevo - anterior;

        let clase = "equal", simbolo = "●", difftxt = "0.00";
        if (diff > 0){
          clase="up"; simbolo="🔺"; difftxt = "+" + diff.toFixed(2);
        }
        if (diff < 0){
          clase="down"; simbolo="🔻"; difftxt = diff.toFixed(2);
        }

        fila.querySelector(".right").innerHTML =
          `${nuevo.toFixed(2)} CUP <span class="${clase}">${simbolo} ${difftxt}</span>`;

        nuevosValores.push({ label: code, valor: nuevo });
        latestRates[code] = nuevo;
      }
    });

    localStorage.setItem("tasas_old", JSON.stringify(tasas));

    if (data.data.date && data.data.hour !== undefined) {
      const hh = String(data.data.hour).padStart(2,'0');
      const mm = String(data.data.minutes).padStart(2,'0');
      const ss = String(data.data.seconds).padStart(2,'0');
      document.getElementById("lastUpdate").innerText =
        `Última actualización: ${data.data.date} ${hh}:${mm}:${ss}`;
    } else {
      document.getElementById("lastUpdate").innerText =
        `Última actualización: datos recibidos correctamente`;
    }

    actualizarGrafica(nuevosValores);
    actualizarConvertidor();

  } catch(e){
    console.error("Error cargando tasas", e);
  }
}

function actualizarGrafica(values){
  const labels = values.map(v => v.label);
  const datos = values.map(v => v.valor);

  const ctx = document.getElementById("graficaMercado").getContext("2d");
  if(chart) chart.destroy();

  chart = new Chart(ctx,{
    type:"bar",
    data:{
      labels,
      datasets:[{
        label:"Tasa en CUP",
        data:datos,
        backgroundColor:"#0b63a5"
      }]
    },
    options:{
      responsive:true,
      plugins:{ legend:{ display:false } }
    }
  });
}

async function cargarDivisasExtra() {
  try {
    const resp = await fetch('divisas_extra.json?' + Date.now(), { cache: "no-store" });
    if (!resp.ok) return;
    const extra = await resp.json();

    const filasFuel = document.querySelectorAll('#combustible-cuba .rate-row[data-fuel-token]');
    filasFuel.forEach(fila => {
      const token = fila.getAttribute('data-fuel-token');
      const info = extra[token];
      if (!info) return;

      const valor   = parseFloat(info.valor);
      const tend    = info.tendencia || 'equal';
      const cambio  = parseFloat(info.cambio || 0);

      if (!isFinite(valor)) return;

      let clase = 'equal';
      let simbolo = '●';
      let diffText = cambio.toFixed(2);

      if (tend === 'up') {
        clase = 'up';
        simbolo = '🔺';
        diffText = (cambio >= 0 ? '+' : '') + cambio.toFixed(2);
      } else if (tend === 'down') {
        clase = 'down';
        simbolo = '🔻';
        diffText = (cambio <= 0 ? '' : '-') + Math.abs(cambio).toFixed(2);
      }

      fila.querySelector('.right').innerHTML =
        `${valor.toFixed(2)} CUP <span class="${clase}">${simbolo} ${diffText}</span>`;
    });

    const fu = document.getElementById('fuelUpdate');
    if (fu) {
      fu.innerText = extra.updated
        ? `Última actualización (combustible): ${extra.updated}`
        : `Última actualización (combustible): datos recibidos`;
    }

  } catch(e) {
    console.error('Error cargando divisas_extra.json', e);
  }
}

async function cargarTasasElToque() {
  try {
    const resp = await fetch('?ajax_eltoque=1&t=' + Date.now(), { cache: "no-store" });
    if (!resp.ok) return;
    const data = await resp.json();

    const map = {
      "CAD": data.CAD,
      "MXN": data.MXN,
      "ZELLE": data.ZELLE
    };

    const filas = document.querySelectorAll('#mercado-informal .rate-row[data-code-eltoque]');
    filas.forEach(fila => {
      const code = fila.getAttribute('data-code-eltoque');
      const val = parseFloat(map[code]);
      if (!isFinite(val)) return;

      const oldRates = JSON.parse(localStorage.getItem('tasas_eltoque_old') || '{}');
      const anterior = oldRates[code] ?? val;
      const diff = val - anterior;

      let clase = 'equal';
      let simbolo = '●';
      let diffText = '0.00';

      if (diff > 0) { 
        clase = 'up'; 
        simbolo = '🔺'; 
        diffText = '+' + diff.toFixed(2); 
      } else if (diff < 0) { 
        clase = 'down'; 
        simbolo = '🔻'; 
        diffText = diff.toFixed(2); 
      }

      fila.querySelector('.right').innerHTML =
        `${val.toFixed(2)} CUP <span class="${clase}">${simbolo} ${diffText}</span>`;

      oldRates[code] = val;
      localStorage.setItem('tasas_eltoque_old', JSON.stringify(oldRates));
    });
  } catch(e) {
    console.error('Error cargando tasas El Toque', e);
  }
}

function actualizarConvertidor(){
  const amountEl = document.getElementById("amount");
  const curEl = document.getElementById("cur");
  const outEl = document.getElementById("convResult");

  if(!amountEl || !curEl || !outEl) return;

  const amount = parseFloat(amountEl.value || "0");
  const cur = curEl.value;

  const rate = latestRates[cur];
  if(!rate || !isFinite(rate) || amount <= 0){
    outEl.textContent = "—";
    return;
  }
  const res = amount * rate;
  outEl.textContent = res.toFixed(2) + " CUP";
}

document.addEventListener("input", (e) => {
  if(e.target && (e.target.id === "amount" || e.target.id === "cur")){
    actualizarConvertidor();
  }
});

async function cargarMercadoFormal(){
  try{
    const r = await fetch("https://api.cambiocuba.money/api/formal_rates/banks-x-rates", { cache:"no-store" });
    const data = await r.json();

    if(!Array.isArray(data) || data.length === 0){
      return;
    }

    const banco = data.find(x => (x?.name || '') === "BANDEC") || data[0];
    const rates = banco?.rates || null;
    if(!rates) return;

    const filas = document.querySelectorAll("#mercado-formal .rate-row[data-formal]");
    filas.forEach(fila => {
      const code = fila.getAttribute("data-formal");
      const info = rates[code];
      if(!info) return;

      const v = (info.sell !== undefined && info.sell !== null) ? parseFloat(info.sell) : parseFloat(info.buy);
      if(!isFinite(v)) return;

      const diff = parseFloat(info.diff || 0);

      let clase="equal", simbolo="●", difftxt="0.00";
      if(diff > 0){
        clase="up"; simbolo="🔺"; difftxt = "+" + diff.toFixed(2);
      } else if(diff < 0){
        clase="down"; simbolo="🔻"; difftxt = diff.toFixed(2);
      }

      fila.querySelector(".right").innerHTML =
        `${v.toFixed(2)} CUP <span class="${clase}">${simbolo} ${difftxt}</span>`;
    });

    const fecha = banco?.effective_date ? banco.effective_date : "";
    const nombre = banco?.name ? banco.name : "Bancos";
    const sub = banco?.subtitle ? (" · " + banco.subtitle) : "";
    const el = document.getElementById("formalUpdate");
    if(el){
      el.innerText = fecha
        ? `Última actualización (formal): ${fecha} · Fuente: ${nombre}${sub}`
        : `Última actualización (formal): datos recibidos · Fuente: ${nombre}${sub}`;
    }

  } catch(e){
    console.error("Error cargando mercado formal", e);
  }
}
cargarMercadoFormal();
setInterval(cargarMercadoFormal, 60000);

(function(){
  const popup = document.getElementById('scRatePopup');
  if (!popup) return;

  const intro = document.getElementById('scRateIntro');
  const form = document.getElementById('scRateForm');
  const msg = document.getElementById('scRateMsg');
  const yesBtn = document.getElementById('scRateYes');
  const noMoreBtn = document.getElementById('scRateNoMore');
  const closeBtn = document.getElementById('scRatePopupClose');
  const backBtn = document.getElementById('scRateBack');
  const submitBtn = document.getElementById('scRateSubmit');

  const STORAGE_KEY = 'sc_rates_popup_done';

  function getCookie(name){
    return document.cookie.split('; ').find(row => row.startsWith(name + '='))?.split('=').slice(1).join('=') || '';
  }

  function setCookie(name, value, days){
    const maxAge = days * 24 * 60 * 60;
    document.cookie = name + '=' + encodeURIComponent(value) + '; path=/; max-age=' + maxAge + '; samesite=lax';
  }

  function popupAlreadyHandled(){
    try {
      if (localStorage.getItem(STORAGE_KEY)) return true;
    } catch(e) {}
    return getCookie('sc_rates_popup_done') === '1';
  }

  function markHandled(value){
    try {
      localStorage.setItem(STORAGE_KEY, value || '1');
    } catch(e) {}
    setCookie('sc_rates_popup_done', '1', 365);
  }

  function showPopup(){
    if (popupAlreadyHandled()) return;
    popup.classList.add('show');
    popup.setAttribute('aria-hidden', 'false');
    document.body.classList.add('sc-popup-open');
  }

  function hidePopup(){
    popup.classList.remove('show');
    popup.setAttribute('aria-hidden', 'true');
    document.body.classList.remove('sc-popup-open');
  }

  function dismissForever(){
    markHandled('dismissed');
    hidePopup();
  }

  window.addEventListener('load', function(){
    setTimeout(showPopup, 1400);
  });

  if (yesBtn) {
    yesBtn.addEventListener('click', function(){
      if (intro) intro.style.display = 'none';
      if (form) form.classList.add('show');
      const nameInput = document.getElementById('scRateName');
      if (nameInput) setTimeout(() => nameInput.focus(), 80);
    });
  }

  [noMoreBtn, closeBtn, backBtn].forEach(function(btn){
    if (!btn) return;
    btn.addEventListener('click', dismissForever);
  });

  popup.addEventListener('click', function(e){
    if (e.target === popup) dismissForever();
  });

  document.addEventListener('keydown', function(e){
    if (e.key === 'Escape' && popup.classList.contains('show')) {
      dismissForever();
    }
  });

  if (form) {
    form.addEventListener('submit', async function(e){
      e.preventDefault();
      if (!msg) return;

      msg.className = 'sc-popup-msg';
      msg.textContent = 'Guardando...';
      if (submitBtn) submitBtn.disabled = true;

      const payload = {
        name: (document.getElementById('scRateName')?.value || '').trim(),
        email: (document.getElementById('scRateEmail')?.value || '').trim(),
        phone: (document.getElementById('scRatePhone')?.value || '').trim(),
        website: (document.getElementById('scRateWebsite')?.value || '').trim()
      };

      try {
        const resp = await fetch(location.pathname + '?sc_register_user=1', {
          method: 'POST',
          headers: {'Content-Type':'application/json'},
          body: JSON.stringify(payload),
          credentials: 'same-origin',
          cache: 'no-store'
        });

        const data = await resp.json().catch(() => ({}));

        if (!resp.ok || !data.ok) {
          throw new Error(data.message || 'No se pudo guardar. Inténtalo otra vez.');
        }

        markHandled('registered');
        msg.className = 'sc-popup-msg ok';
        msg.textContent = data.message || 'Cuenta creada correctamente.';
        setTimeout(hidePopup, 1300);
      } catch(err) {
        msg.className = 'sc-popup-msg err';
        msg.textContent = err.message || 'Error guardando los datos.';
      } finally {
        if (submitBtn) submitBtn.disabled = false;
      }
    });
  }
})();
